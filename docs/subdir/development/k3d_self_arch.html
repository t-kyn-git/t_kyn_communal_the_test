PlantUML Code


@startuml
!theme cerulean-outline

actor Client

cloud "GitHub Codespaces" {
    folder "Host Path Volume" as HostVolume
    
    package "k3d Cluster" {
        node "Load Balancer" as LB

        package "Worker Nodes" {
            rectangle "Nginx Pods" as Nginx {
                component [nginx-1]
                component [nginx-2]
                component [nginx-3]
            }
            rectangle "Python Pods (FastAPI)" as Python {
                component [python-1]
                component [python-2]
                component [python-3]
            }
            database "MySQL Pods" as MySQL {
                database "mysql-master" as Master
                database "mysql-slave-1" as Slave1
                database "mysql-slave-2" as Slave2
            }
            
            package "Monitoring & Logging" {
                component [Prometheus]
                component [Grafana]
                component [Fluentd]
            }
        }

        package "Kubernetes Services" {
            hexagon "nginx-svc" as NginxSvc
            hexagon "python-svc" as PythonSvc
            hexagon "mysql-write-svc" as MySQLWriteSvc
            hexagon "mysql-read-svc" as MySQLReadSvc
        }

        package "Storage" {
            storage "PVCs" as PVC
        }
    }
}

' --- Connections ---
Client -down-> LB : HTTPS
LB -down-> NginxSvc

NginxSvc -right-> Nginx
Nginx -down-> PythonSvc
PythonSvc -right-> Python

Python -down-> MySQLWriteSvc : Write
Python -down-> MySQLReadSvc : Read

MySQLWriteSvc -right-> Master
MySQLReadSvc -right-> Master
MySQLReadSvc -right-> Slave1
MySQLReadSvc -right-> Slave2

MySQL -down-> PVC
PVC -down- HostVolume : Binds

' --- Monitoring ---
Prometheus .up.> Nginx : Scrapes
Prometheus .up.> Python : Scrapes
Prometheus .up.> MySQL : Scrapes
Grafana .left.> Prometheus : Queries
Client -right-> Grafana : Views Dashboard

@enduml
